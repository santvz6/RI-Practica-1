# ==============================================================================
# 1. COMPILACIÓN DE LA IMAGEN DOCKER
# ==============================================================================
docker build -t robotica_inteligente_grado:latest .

# ==============================================================================
# 2. CREACIÓN Y ARRANQUE DEL CONTENEDOR (CON ACELERACIÓN NVIDIA GPU)
# ==============================================================================
# Permitir acceso al servidor X11 local
xhost +local:docker

# Opción A: Arrancar en segundo plano (Daemon) para abrir terminales múltiples
docker run -d \
  --user developer \
  --network host \
  --ipc host \
  --pid host \
  --env DISPLAY=$DISPLAY \
  --env QT_X11_NO_MITSHM=1 \
  --env NVIDIA_VISIBLE_DEVICES=all \
  --env NVIDIA_DRIVER_CAPABILITIES=all \
  --env __NV_PRIME_RENDER_OFFLOAD=1 \
  --env __GLX_VENDOR_LIBRARY_NAME=nvidia \
  --env __GL_THREADED_OPTIMIZATIONS=0 \
  --volume /tmp/.X11-unix:/tmp/.X11-unix:rw \
  --volume ~/ros2_ws:/home/developer/ros2_ws:rw \
  --device /dev/dri \
  --gpus all \
  --runtime nvidia \
  --name robotica_inteligente_grado \
  robotica_inteligente_grado:latest \
  sleep infinity
